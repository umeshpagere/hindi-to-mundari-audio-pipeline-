#!/usr/bin/env python3
import collections
import os
from typing import Any, Dict

import onnx
import torch
from vits import commons, utils
from vits.models import SynthesizerTrn


class OnnxModel(torch.nn.Module):
    def __init__(self, model: SynthesizerTrn):
        super().__init__()
        self.model = model

    def forward(self, x, x_lengths, noise_scale=0.667, length_scale=1.0, noise_scale_w=0.8):
        return self.model.infer(
            x=x, x_lengths=x_lengths,
            noise_scale=noise_scale, length_scale=length_scale, noise_scale_w=noise_scale_w,
        )[0]


def add_meta_data(filename: str, meta_data: Dict[str, Any]):
    model = onnx.load(filename)
    for key, value in meta_data.items():
        meta = model.metadata_props.add()
        meta.key = key
        meta.value = str(value)
    onnx.save(model, filename)


def load_vocab():
    return [x.replace("\n", "") for x in open("vocab.txt", encoding="utf-8").readlines()]


@torch.no_grad()
def main():
    hps = utils.get_hparams_from_file("config.json")
    is_uroman = hps.data.training_files.split(".")[-1] == "uroman"
    if is_uroman:
        raise ValueError("We don't support uroman!")

    symbols = load_vocab()
    all_upper_tokens = [i.upper() for i in symbols]
    duplicate = set(item for item, count in collections.Counter(all_upper_tokens).items() if count > 1)

    print("generate tokens.txt")
    with open("tokens.txt", "w", encoding="utf-8") as f:
        for idx, token in enumerate(symbols):
            f.write(f"{token} {idx}\n")
            if token.lower() != token.upper() and len(token.upper()) == 1 and token.upper() not in duplicate:
                f.write(f"{token.upper()} {idx}\n")

    net_g = SynthesizerTrn(
        len(symbols),
        hps.data.filter_length // 2 + 1,
        hps.train.segment_size // hps.data.hop_length,
        **hps.model,
    )
    net_g.cpu()
    _ = net_g.eval()
    _ = utils.load_checkpoint("G_100000.pth", net_g, None)

    model = OnnxModel(net_g)

    x = torch.randint(low=1, high=10, size=(50,), dtype=torch.int64).unsqueeze(0)
    x_length = torch.tensor([x.shape[1]], dtype=torch.int64)
    noise_scale = torch.tensor([1], dtype=torch.float32)
    length_scale = torch.tensor([1], dtype=torch.float32)
    noise_scale_w = torch.tensor([1], dtype=torch.float32)

    filename = "model.onnx"
    torch.onnx.export(
        model, (x, x_length, noise_scale, length_scale, noise_scale_w), filename,
        opset_version=13,
        input_names=["x", "x_length", "noise_scale", "length_scale", "noise_scale_w"],
        output_names=["y"],
        dynamic_axes={"x": {0: "N", 1: "L"}, "x_length": {0: "N"}, "y": {0: "N", 2: "L"}},
    )
    meta_data = {
        "model_type": "vits",
        "comment": "mms",
        "url": "https://huggingface.co/facebook/mms-tts/tree/main",
        "add_blank": int(hps.data.add_blank),
        "language": os.environ.get("language", "unknown"),
        "frontend": "characters",
        "n_speakers": int(hps.data.n_speakers),
        "sample_rate": hps.data.sampling_rate,
    }
    print("meta_data", meta_data)
    add_meta_data(filename=filename, meta_data=meta_data)


main()

hps = utils.get_hparams_from_file("config.json")

# Load vocabulary
with open("vocab.txt", "r", encoding="utf-8") as f:
    symbols = [line.strip() for line in f.readlines()]
symbol_to_id = {s: i for i, s in enumerate(symbols)}

# Initialize and load weights
net_g = SynthesizerTrn(
    len(symbols),
    hps.data.filter_length // 2 + 1,
    hps.train.segment_size // hps.data.hop_length,
    **hps.model,
)

# Helper function to convert string text into PyTorch tensors
def text_to_sequence(text):
    seq = [symbol_to_id[char] for char in text if char in symbol_to_id]
    if hps.data.add_blank:
        # VITS often interleaves a blank token (usually 0) between characters
        seq_with_blanks = [0]
        for s in seq:
            seq_with_blanks.extend([s, 0])
        seq = seq_with_blanks
    return torch.LongTensor(seq).unsqueeze(0)

# 2. Setup tracking variables
process = psutil.Process(os.getpid())
latencies_pt = []
ram_usages_mb_pt = []

print("\nStarting Performance Benchmark for Native PyTorch Model...\n")

# Optional warmup run

    _x = text_to_sequence("Warmup")
    _x_lengths = torch.LongTensor([_x.size(1)])
    _ = net_g.infer(_x, _x_lengths, noise_scale=0.667, length_scale=1.0, noise_scale_w=0.8)

import onnx
from collections import defaultdict

model_path = "model.onnx"
onnx_model = onnx.load(model_path)

# Build a map of which node produces each tensor
producer_map = {}
for node in onnx_model.graph.node:
    for output in node.output:
        producer_map[output] = node

# Trace backwards from the final output to find dependencies
output_tensor = onnx_model.graph.output[0].name  # should be 'y'
print(f"Output tensor: {output_tensor}")

# Follow the graph backwards
visited = set()
stack = [output_tensor]
depth = 0
max_depth = 15  # limit backward search

print("\n=== Graph trace from output (backwards, first 15 steps) ===")
while stack and depth < max_depth:
    tensor = stack.pop(0)
    if tensor in visited:
        continue
    visited.add(tensor)
    
    if tensor not in producer_map:
        print(f"[{depth:2d}] (input) {tensor}")
        depth += 1
        continue
    
    node = producer_map[tensor]
    print(f"[{depth:2d}] {node.op_type:15s} -> {tensor[:40]}")
    
    # Add inputs to stack for further tracing
    for inp in node.input:
        if inp not in visited:
            stack.append(inp)
    
    depth += 1

print("\n=== ConvTranspose nodes (vocoder upsampling) ===")
for i, node in enumerate(onnx_model.graph.node):
    if node.op_type == "ConvTranspose":
        print(f"  [{i:4d}] output: {node.output[0]}")

print("\n=== Last 50 nodes (likely vocoder) ===")
for i, node in enumerate(onnx_model.graph.node[-50:], start=len(onnx_model.graph.node)-50):
    print(f"[{i:4d}] {node.op_type:15s} -> {node.output[0][:50]}")

class VitsCalibrationReader(CalibrationDataReader):
    def __init__(self, text_list, tokenizer_path):
        self.data = []
        
        # Load token map from tokens.txt
        token_map = {}
        with open(tokenizer_path, encoding='utf-8') as f:
            for line in f:
                parts = line.rsplit(' ', 1)
                if len(parts) == 2:
                    token_map[parts[0]] = int(parts[1])
        
        # Convert text to token IDs
        for text in text_list:
            token_ids = []
            for c in text:
                if c != ' ' and c in token_map:
                    token_ids.append(token_map[c])
            
            if token_ids:
                self.data.append(token_ids)
        
        self.index = 0
        print(f"Calibration reader initialized with {len(self.data)} sequences")

    def get_next(self):
        if self.index >= len(self.data):
            return None
        
        token_ids = self.data[self.index]
        # Prepare inputs matching ONNX model signature
        x = np.array([token_ids], dtype=np.int64)  # shape: (1, seq_len)
        x_length = np.array([len(token_ids)], dtype=np.int64)
        noise_scale = np.array([0.667], dtype=np.float32)
        length_scale = np.array([1.0], dtype=np.float32)
        noise_scale_w = np.array([0.8], dtype=np.float32)
        
        self.index += 1
        
        return {
            "x": x,
            "x_length": x_length,
            "noise_scale": noise_scale,
            "length_scale": length_scale,
            "noise_scale_w": noise_scale_w,
        }

# Rebuild compute_mcd functions locally in the notebook
import numpy as np
from scipy.io import wavfile
from scipy.fftpack import dct

def hz_to_mel(hz):
    return 2595.0 * np.log10(1.0 + hz / 700.0)

def mel_to_hz(mel):
    return 700.0 * (10.0 ** (mel / 2595.0) - 1.0)

def mel_filterbank(sr, n_fft, n_mels=26, fmin=0, fmax=None):
    fmax = fmax or sr / 2
    mel_min, mel_max = hz_to_mel(fmin), hz_to_mel(fmax)
    mel_points = np.linspace(mel_min, mel_max, n_mels + 2)
    hz_points = mel_to_hz(mel_points)
    bins = np.floor((n_fft + 1) * hz_points / sr).astype(int)

    fbank = np.zeros((n_mels, n_fft // 2 + 1))
    for m in range(1, n_mels + 1):
        f_left, f_center, f_right = bins[m - 1], bins[m], bins[m + 1]
        for k in range(f_left, f_center):
            if f_center != f_left:
                fbank[m - 1, k] = (k - f_left) / (f_center - f_left)
        for k in range(f_center, f_right):
            if f_right != f_center:
                fbank[m - 1, k] = (f_right - k) / (f_right - f_center)
    return fbank

def compute_mfcc(wav_path, n_mfcc=13, n_mels=26, frame_len_ms=25, frame_shift_ms=10):
    sr, data = wavfile.read(wav_path)
    if data.dtype == np.int16:
        data = data.astype(np.float32) / 32768.0
    elif data.dtype == np.int32:
        data = data.astype(np.float32) / 2147483648.0
    else:
        data = data.astype(np.float32)
    if data.ndim > 1:
        data = data.mean(axis=1)

    frame_len = int(sr * frame_len_ms / 1000)
    frame_shift = int(sr * frame_shift_ms / 1000)
    n_fft = 1
    while n_fft < frame_len:
        n_fft *= 2

    n_frames = 1 + max(0, (len(data) - frame_len) // frame_shift)
    if n_frames <= 0:
        return np.zeros((0, n_mfcc)), sr

    window = np.hanning(frame_len)
    fbank = mel_filterbank(sr, n_fft, n_mels=n_mels)

    mfccs = []
    for i in range(n_frames):
        start = i * frame_shift
        frame = data[start:start + frame_len]
        if len(frame) < frame_len:
            frame = np.pad(frame, (0, frame_len - len(frame)))
        frame = frame * window
        spectrum = np.abs(np.fft.rfft(frame, n=n_fft)) ** 2
        mel_energy = fbank @ spectrum
        mel_energy = np.where(mel_energy == 0, np.finfo(float).eps, mel_energy)
        log_mel = np.log(mel_energy)
        mfcc = dct(log_mel, type=2, norm="ortho")[:n_mfcc]
        mfccs.append(mfcc)

    return np.array(mfccs), sr

def dtw_align(a, b):
    n, m = len(a), len(b)
    if n == 0 or m == 0:
        return []
    cost = np.zeros((n, m))
    for i in range(n):
        diff = a[i] - b
        cost[i] = np.sqrt(np.sum(diff ** 2, axis=1))

    acc = np.full((n, m), np.inf)
    acc[0, 0] = cost[0, 0]
    for i in range(1, n):
        acc[i, 0] = acc[i - 1, 0] + cost[i, 0]
    for j in range(1, m):
        acc[0, j] = acc[0, j - 1] + cost[0, j]
    for i in range(1, n):
        for j in range(1, m):
            acc[i, j] = cost[i, j] + min(acc[i - 1, j], acc[i, j - 1], acc[i - 1, j - 1])

    i, j = n - 1, m - 1
    pairs = [(i, j)]
    while i > 0 or j > 0:
        if i == 0:
            j -= 1
        elif j == 0:
            i -= 1
        else:
            choices = [acc[i - 1, j - 1], acc[i - 1, j], acc[i, j - 1]]
            k = np.argmin(choices)
            if k == 0:
                i, j = i - 1, j - 1
            elif k == 1:
                i -= 1
            else:
                j -= 1
        pairs.append((i, j))
    pairs.reverse()
    return pairs

def mcd_between(path_a, path_b):
    mfcc_a, _ = compute_mfcc(path_a)
    mfcc_b, _ = compute_mfcc(path_b)
    mfcc_a, mfcc_b = mfcc_a[:, 1:], mfcc_b[:, 1:]
    pairs = dtw_align(mfcc_a, mfcc_b)
    diffs = []
    for i, j in pairs:
        d = mfcc_a[i] - mfcc_b[j]
        diffs.append(np.sqrt(np.sum(d ** 2)))
    diffs = np.array(diffs)
    K = 10.0 / np.log(10) * np.sqrt(2)
    return K * np.mean(diffs)

class VitsCalibrationReader(CalibrationDataReader):
    def __init__(self, text_list, tokenizer_path):
        self.data = []
        token_map = {}
        with open(tokenizer_path, encoding='utf-8') as f:
            for line in f:
                parts = line.rsplit(' ', 1)
                if len(parts) == 2:
                    token_map[parts[0]] = int(parts[1])
        
        for text in text_list:
            token_ids = [token_map.get(c, 0) for c in text if c != ' ' and c in token_map]
            if token_ids:
                self.data.append(token_ids)
        
        self.index = 0

    def get_next(self):
        if self.index >= len(self.data):
            return None
        
        token_ids = self.data[self.index]
        x = np.array([token_ids], dtype=np.int64)
        x_length = np.array([len(token_ids)], dtype=np.int64)

# Strategy: Keep everything high-precision except duration predictor / length regulator
# Identify text encoder nodes (early layers, before duration predictor)
# and vocoder nodes (late layers, /dec/), exclude both from INT8

EXCLUDE_NODES_AGGRESSIVE = []

# Exclude all /dec/ (vocoder) nodes
for node in onnx_model.graph.node:
    for output in node.output:
        if "/dec/" in output:
            EXCLUDE_NODES_AGGRESSIVE.append(output)
            break

# Also exclude early text encoder layers (roughly first 30% of nodes)
# These handle character embeddings and text representation
early_encoder_cutoff = int(len(onnx_model.graph.node) * 0.30)
for node in onnx_model.graph.node[:early_encoder_cutoff]:
    for output in node.output:
        EXCLUDE_NODES_AGGRESSIVE.append(output)

class VitsCalibrationReader(CalibrationDataReader):
    def __init__(self, text_list, tokenizer_path):
        self.data = []
        token_map = {}
        with open(tokenizer_path, encoding='utf-8') as f:
            for line in f:
                parts = line.rsplit(' ', 1)
                if len(parts) == 2:
                    token_map[parts[0]] = int(parts[1])
        
        for text in text_list:
            token_ids = [token_map.get(c, 0) for c in text if c != ' ' and c in token_map]
            if token_ids:
                self.data.append(token_ids)
        
        self.index = 0

    def get_next(self):
        if self.index >= len(self.data):
            return None
        
        token_ids = self.data[self.index]
        x = np.array([token_ids], dtype=np.int64)
        x_length = np.array([len(token_ids)], dtype=np.int64)
        noise_scale = np.array([0.667], dtype=np.float32)
        length_scale = np.array([1.0], dtype=np.float32)
        noise_scale_w = np.array([0.8], dtype=np.float32)
        
        self.index += 1
        
        return {
            "x": x,
            "x_length": x_length,
            "noise_scale": noise_scale,
            "length_scale": length_scale,
            "noise_scale_w": noise_scale_w,
        }

# ── Mundari INT8-Dynamic ONNX TTS — Single-Cell Test ─────────────────────────
import os
import numpy as np
import onnxruntime as rt
import scipy.io.wavfile as wavfile
from IPython.display import Audio, display

# ── 1. Config (matches your config.json) ─────────────────────────────────────
MODEL_PATH   = "model_int8_dynamic.onnx"
TOKENS_PATH  = "tokens.txt"
SAMPLE_RATE  = 16000   # config.json → data.sampling_rate
ADD_BLANK    = True    # config.json → data.add_blank
NOISE_SCALE  = 0.667
LENGTH_SCALE = 1.0
NOISE_SCALE_W = 0.8

# ── 2. Load vocabulary from tokens.txt ───────────────────────────────────────
char2id = {}
with open(TOKENS_PATH, encoding="utf-8") as f:
    for line in f:
        line = line.rstrip("\n")
        if not line:
            continue
        parts = line.rsplit(" ", 1)
        if len(parts) == 2:
            char2id[parts[0]] = int(parts[1])

print(f"✅ Vocabulary loaded: {len(char2id)} tokens")
print("   Known chars:", list(char2id.keys())[:10], "...")

# ── 3. Text → token IDs (with optional blank interleaving) ───────────────────
def text_to_ids(text: str, vocab: dict, add_blank: bool = True) -> list[int]:
    BLANK_ID = 0           # blank token appended after known vocab
    ids = []
    skipped = []
    for ch in text:
        if ch in vocab:
            ids.append(vocab[ch])
        else:
            skipped.append(repr(ch))
    if skipped:
        print(f"⚠️  Skipped {len(skipped)} unknown chars: {set(skipped)}")
    if add_blank:
        # Interleave blank: [blank, t0, blank, t1, blank, ...]
        result = []
        for tid in ids:
            result.append(BLANK_ID)
            result.append(tid)
        result.append(BLANK_ID)
        return result
    return ids

# ── 4. Test sentences (Odia script Mundari) ──────────────────────────────────
test_sentences = [
    "ଦୁବଇ ଏଯର ପୋଟ କେ ରନବୀ ପର  ସ୍ଟାଇ ଜେଟ କୀ ଫ୍ଲାଇଟ  ଉଡାନ ବରନେ କୋ ପିଲ କୋଲତା ଯାରତୀ  ସଭୀ ପେସିଂଜେସ ଫଲାଇଟ କୀ"               # "Namaste / Hello"
]

# ── 5. Load ONNX model ────────────────────────────────────────────────────────
print(f"\n📦 Loading: {MODEL_PATH}  ({os.path.getsize(MODEL_PATH)/1e6:.1f} MB)")
sess_opts = rt.SessionOptions()
sess_opts.graph_optimization_level = rt.GraphOptimizationLevel.ORT_ENABLE_ALL
session = rt.InferenceSession(MODEL_PATH, sess_options=sess_opts,
                              providers=["CPUExecutionProvider"])

input_names  = [i.name for i in session.get_inputs()]
output_names = [o.name for o in session.get_outputs()]
print(f"✅ Model loaded   | Inputs: {input_names}  | Outputs: {output_names}")

# ── 6. Synthesize each test sentence ─────────────────────────────────────────
for idx, text in enumerate(test_sentences):
    print(f"\n── [{idx+1}/{len(test_sentences)}] Synthesizing: '{text}' ──")

    ids = text_to_ids(text, char2id, add_blank=ADD_BLANK)
    if not ids:
        print("   ❌ No valid tokens — skipping.")
        continue

    x        = np.array(ids, dtype=np.int64)[np.newaxis, :]       # (1, L)
    x_length = np.array([x.shape[1]], dtype=np.int64)              # (1,)
    noise    = np.array([NOISE_SCALE],   dtype=np.float32)
    length   = np.array([LENGTH_SCALE],  dtype=np.float32)
    noise_w  = np.array([NOISE_SCALE_W], dtype=np.float32)

    feeds = {
        "x":             x,
        "x_length":      x_length,
        "noise_scale":   noise,
        "length_scale":  length,
        "noise_scale_w": noise_w,
    }

    audio = session.run(output_names, feeds)[0]          # (1, 1, T)
    audio = audio.squeeze()                               # (T,)

    # Normalise to int16 range for WAV
    audio_norm = audio / (np.abs(audio).max() + 1e-9)
    audio_int16 = (audio_norm * 32767).astype(np.int16)

    out_path = f"test_output_{idx+1}.wav"
    wavfile.write(out_path, SAMPLE_RATE, audio_int16)

    duration_s = len(audio) / SAMPLE_RATE
    print(f"   ✅ {duration_s:.2f}s  →  saved: {out_path}")
    display(Audio(audio_norm, rate=SAMPLE_RATE, autoplay=False))

print("\n🎉 All done!")
